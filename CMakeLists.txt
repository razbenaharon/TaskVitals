cmake_minimum_required(VERSION 3.20)
project(TaskVitals VERSION 1.9.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TaskVitals requires Windows")
endif()
add_executable(TaskVitals WIN32 src/main.cpp src/gpu.h src/thermal.h src/network.h src/TaskVitals.rc)
target_compile_features(TaskVitals PRIVATE cxx_std_11)
target_link_libraries(TaskVitals PRIVATE user32 gdi32 advapi32 comctl32 imm32 oleacc ole32 oleaut32 shell32 uuid)
if(MSVC)
  target_compile_options(TaskVitals PRIVATE /W4 /utf-8 /O2)
  set_property(TARGET TaskVitals PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(TaskVitals PRIVATE -Wall -Wextra -O2 -finput-charset=UTF-8)
  target_link_options(TaskVitals PRIVATE -static -static-libgcc -static-libstdc++ -s)
endif()
enable_testing()
add_test(NAME sampling-self-test COMMAND TaskVitals --self-test)
